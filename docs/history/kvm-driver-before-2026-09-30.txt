#!/bin/sh
# Question: does the ValidationOS guest run under kvm from this tree's
# committed verbs, with the acceleration measured from outside?
#
# TODO/milestones.md T-1112. The TCG arms are proved; the kvm arm of
# `accel_for` was unit-tested only, and the host KVM node answers API
# version 12 (`experiments/385-kvm-open.sh`). This drive runs the guest
# where the node lives: the wsl-toolkit base holds QEMU 11.1.1 with kvm
# in its accelerator list and opens /dev/kvm, so the run executes there
# via `base exec` while the verdict lands here. `podbox windows doctor`
# proves profile full with the kvm accelerator, `setup` provisions the
# agent into a scratch copy of the installed ValidationOS disk (read
# once from the host automount, verified against the pinned digest,
# never committed or redistributed; the operator accepts the image
# terms as standing policy since 2026-09-28), and `run` returns the
# guest's version string plus exit 42 through the committed verbs and
# through the `podbox run` seam. An outside observer captures the
# emulator command line during the run and asserts `-accel kvm
# -cpu host`, so the acceleration is measured, not trusted from the
# driver's own report.
#
# Pinned inputs: the lane-built podbox binary (musl, from
# `cargo build -p podbox-cli --target x86_64-unknown-linux-musl`,
# staged at `.tmp/podbox-kvm/podbox`, refuse without it), the
# installed ValidationOS disk (910163968 bytes, sha256
# checksum: 063442aa9f71f2faeebf49cd960003ce315abd556b52c696e1b994ec5a80fe7f
# from `experiments/371-validationos-stream.sh`), edk2-ovmf 4M pair in
# the base. Conditions print on the way out.
#
# Exit 0 the guest ran under kvm with its version and exit codes,
# 1 it ran and disagreed, 2 the lane could not run it.
set -u

HERE=$(CDPATH= cd -- "$(dirname -- "$0")" && pwd) || exit 2
ROOT=$(CDPATH= cd -- "$HERE/.." && pwd) || exit 2
OUT="$ROOT/experiments/results/kvm-guest.txt"

if ! command -v wsl-toolkit >/dev/null 2>&1; then
  echo "kvm-guest: wsl-toolkit is not on PATH" >&2
  exit 2
fi
if [ ! -s "$ROOT/.tmp/podbox-kvm/podbox" ]; then
  echo "kvm-guest: no lane-built binary at .tmp/podbox-kvm/podbox" >&2
  echo "  build it first: cargo build -p podbox-cli --target x86_64-unknown-linux-musl" >&2
  exit 2
fi
work=$(mktemp -d) || exit 2
trap 'rm -rf "$work"' EXIT HUP INT TERM

cat >"$work/base-run.sh" <<'BASE'
#!/bin/sh
# T-1112 KVM guest run, executed in the wsl-toolkit base (the KVM host).
set -u

PODBOX_BIN_SRC="$HOST_INPUT"
KVM="$HOME/podbox-kvm"
PODBOX="$KVM/podbox"
IMG_HOST="$HOST_INPUT"
VHDX_LEN=910163968
VHDX_SHA256="063442aa9f71f2faeebf49cd960003ce315abd556b52c696e1b994ec5a80fe7f"
CODE="/usr/share/edk2-ovmf/x64/OVMF_CODE.4m.fd"
VARS="/usr/share/edk2-ovmf/x64/OVMF_VARS.4m.fd"

fail=0
say() { printf '%s\n' "$*"; }
miss() { fail=1; say "FAIL: $1"; }
ok() { say "ok: $1"; }

mkdir -p "$KVM" || { miss "cannot create $KVM"; printf 'verdict KVM-GUEST-FAIL\n'; exit 1; }
cp "$PODBOX_BIN_SRC" "$PODBOX" && chmod +x "$PODBOX" \
  || { miss "cannot stage the podbox binary"; printf 'verdict KVM-GUEST-FAIL\n'; exit 1; }
[ -x "$PODBOX" ] || { miss "staged binary not executable"; printf 'verdict KVM-GUEST-FAIL\n'; exit 1; }
export PODBOX_STORE="$KVM/store"
export PODBOX_OVMF_CODE="$CODE" PODBOX_OVMF_VARS="$VARS"
mkdir -p "$PODBOX_STORE" || exit 1

say "== conditions"
date -u +%Y-%m-%dT%H:%M:%SZ
uname -sr
qemu-system-x86_64 --version | head -n 1
qemu-system-x86_64 -accel help 2>/dev/null | tr '\n' ' '
echo
"$PODBOX" --version 2>/dev/null || "$PODBOX" version 2>/dev/null || echo "podbox binary answers"
free -m | head -n 2
ls -l /dev/kvm /dev/net/tun
echo

say "== image"
if [ -f "$IMG_HOST" ]; then
  say "host image bytes: $(stat -c %s "$IMG_HOST" 2>/dev/null || stat -f %z "$IMG_HOST")"
else
  miss "host image absent at $IMG_HOST"
fi
if [ "$fail" -eq 0 ]; then
  cp "$IMG_HOST" "$KVM/base.vhdx" || miss "image copy failed"
  got=$(stat -c %s "$KVM/base.vhdx" 2>/dev/null || stat -f %z "$KVM/base.vhdx")
  [ "$got" = "$VHDX_LEN" ] || miss "image length $got, not $VHDX_LEN"
  sum=$(sha256sum "$KVM/base.vhdx" 2>/dev/null | cut -d' ' -f1)
  [ "$sum" = "$VHDX_SHA256" ] || miss "image digest mismatch"
  [ "$fail" -eq 0 ] && ok "installed disk copied with the pinned digest"
fi

say "== doctor"
if [ "$fail" -eq 0 ]; then
  if timeout 120 "$PODBOX" windows doctor >"$KVM/doctor.txt" 2>&1; then
    cat "$KVM/doctor.txt"
    if grep -qi "accelerator: kvm" "$KVM/doctor.txt"; then
      ok "doctor names the kvm accelerator"
    else
      miss "doctor does not name kvm"
    fi
  else
    miss "doctor refused"; cat "$KVM/doctor.txt"
  fi
fi

say "== setup"
PROV="$KVM/base.podbox.qcow2"
# A re-drive provisions again from the verified base rather than
# reusing the last run's image: the provision is part of the proof.
rm -f "$PROV"
if [ "$fail" -eq 0 ]; then
  if timeout 900 "$PODBOX" windows setup --image "$KVM/base.vhdx" --podbox-timeout 600 >"$KVM/setup.txt" 2>&1; then
    tail -n 4 "$KVM/setup.txt"
    [ -f "$PROV" ] && ok "provisioned image beside the base" || miss "provisioned image absent"
  else
    miss "setup failed"; tail -n 10 "$KVM/setup.txt"
  fi
fi

say "== run ver with the emulator command line watched"
if [ "$fail" -eq 0 ]; then
  timeout 600 "$PODBOX" windows run --image "$PROV" --podbox-timeout 540 -- ver >"$KVM/ver.txt" 2>"$KVM/ver.err" &
  runpid=$!
  qemu_cmd=""
  waited=0
  while [ "$waited" -lt 540 ]; do
    if ! kill -0 "$runpid" 2>/dev/null; then break; fi
    qemu_cmd=$(ps -ef 2>/dev/null | grep "[q]emu-system-x86_64" | head -n 1)
    [ -n "$qemu_cmd" ] && break
    sleep 2
    waited=$((waited + 2))
  done
  wait "$runpid"
  vcode=$?
  say "run exit: $vcode"
  # The transcript stays reviewable: cmd.exe writes CRLF and a raw byte
  # would make this file invisible to grep, so the asserts above read
  # the raw capture and only the printed copy is scrubbed.
  tr -d '\r' <"$KVM/ver.txt"
  say "emulator: $qemu_cmd"
  if [ "$vcode" -eq 0 ] && grep -q "Microsoft Windows" "$KVM/ver.txt"; then
    ok "guest version string with exit 0"
  else
    miss "ver run failed"; tail -n 5 "$KVM/ver.err"
  fi
  case "$qemu_cmd" in
    *"-accel kvm"*"-cpu host"*) ok "outside observer: the guest ran under -accel kvm -cpu host" ;;
    *) miss "emulator command line does not show kvm" ;;
  esac
fi

say "== run exit 42"
if [ "$fail" -eq 0 ]; then
  # A serial watcher runs beside the guest: one 540 s hang on
  # 2026-09-29 never reproduced, and the driver's scratch cleanup
  # takes the serial log with it, so the watch copy survives to say
  # where a future hang stops.
  (n=0; while [ "$n" -lt 55 ]; do sleep 10; n=$((n + 1));
    for d in /mnt/wslg/runtime-dir/podbox-windows-*-run "${XDG_RUNTIME_DIR:-/tmp}"/podbox-windows-*-run /tmp/podbox-windows-*-run; do
      [ -f "$d/serial.log" ] && cp "$d/serial.log" "$KVM/serial-watch.log" 2>/dev/null
    done; done) &
  watchpid=$!
  timeout 600 "$PODBOX" windows run --image "$PROV" --podbox-timeout 540 -- ver '>nul' '&' cmd /c exit 42 >"$KVM/e42.txt" 2>"$KVM/e42.err"
  ecode=$?
  kill "$watchpid" 2>/dev/null || true
  say "run exit: $ecode"
  if [ "$ecode" -eq 42 ]; then
    ok "the guest's own exit 42 passed through"
  else
    miss "exit 42 failed with $ecode"; tail -n 5 "$KVM/e42.err"
  fi
fi

say "== the podbox run seam"
if [ "$fail" -eq 0 ]; then
  if timeout 600 "$PODBOX" run --podbox-tier=machine --platform windows/amd64 "$PROV" cmd /c ver >"$KVM/seam.txt" 2>"$KVM/seam.err"; then
    scode=$?
    tr -d '\r' <"$KVM/seam.txt"
    if grep -q "Microsoft Windows" "$KVM/seam.txt"; then
      ok "podbox run reached the disk guest with exit $scode"
    else
      miss "seam run missed the version string"
    fi
  else
    miss "podbox run seam failed"; tail -n 5 "$KVM/seam.err"
  fi
fi

say "== residue"
left=$(ls -d "${XDG_RUNTIME_DIR:-/tmp}"/podbox-windows-*-run /tmp/podbox-windows-*-run 2>/dev/null || true)
if [ -z "$left" ]; then
  ok "no per-run scratch directory remains"
else
  miss "scratch residue: $left"
fi
ls "$KVM"
echo
if [ "$fail" -eq 0 ]; then
  echo "verdict KVM-GUEST-OK: ValidationOS under kvm returns its version and its exit codes"
else
  echo "verdict KVM-GUEST-FAIL"
fi
exit "$fail"
BASE

mkdir -p "$ROOT/experiments/results" || exit 2
{
  echo "== conditions"
  date -u +%Y-%m-%dT%H:%M:%SZ
  timeout 1m wsl-toolkit --version
  git -C "$ROOT" rev-parse --short HEAD
  echo "instance=podbox"
  echo
  echo "== kvm guest run"
} >"$OUT"

timeout 60m wsl-toolkit --instance podbox base exec --timeout 55m --script "$work/base-run.sh" >"$work/run.out" 2>&1
rc=$?
grep -av "instance podbox: distribution" "$work/run.out" >>"$OUT" || true

{
  echo
  if [ "$rc" -eq 0 ] && grep -q "verdict KVM-GUEST-OK" "$OUT"; then
    echo "verdict           KVM-GUEST-OK recorded"
  else
    echo "verdict           KVM-GUEST-FAIL (rc=$rc)"
  fi
} >>"$OUT"
cat "$OUT"
[ "$rc" -eq 0 ] && grep -q "verdict KVM-GUEST-OK" "$OUT"
